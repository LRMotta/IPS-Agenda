function telefoneDe(lab) {
  const t = String(lab.telefone || '').trim();
  if (!t) return { ddd: '-', numero: '-', falta: true };
  const m = t.match(/^(\+\d{1,3})[\s-]*(.+)$/);
  if (m) return { ddd: m[1], numero: m[2].trim() };
  return { ddd: DDI[String(lab.pais || '').trim().toLowerCase()] || '-', numero: t };
}
function cidadeDe(texto) {
  const s = String(texto || '').trim(), i = s.lastIndexOf(',');
  return i < 0 ? { municipio: s, uf: '-' } : { municipio: s.slice(0, i).trim(), uf: s.slice(i + 1).trim() };
}
function volume(ml) { return Number(ml).toFixed(2).replace('.', ','); }
function linhasMaterial(materiais) {
  return materiais.map((m, i) => ('0' + (i + 1)).slice(-2) + '. MATERIAL BIOLÓGICO HUMANO. ' + String(m.tipo).toUpperCase() + '. QUANTIDADE: ' + volume(m.totalMl) + ' mL');
}
function dados(p) {
  const avisos = [], erros = [];
  if (!String(p.awb || '').trim()) erros.push('Informe a AWB antes de gerar.');
  if (!String(p.numeroCE || '').trim()) erros.push('O estudo não tem N° do CE no cadastro.');
  if (p.materiais.length > 6) erros.push('O formulário aceita até 6 materiais.');
  if (!String(p.courier.empresa || '').trim() || !String(p.courier.cnpj || '').trim()) erros.push('Courier sem razão social ou CNPJ no cadastro.');
  const tel = telefoneDe(p.lab);
  if (tel.falta) avisos.push('Laboratório sem telefone no cadastro.');
  const cep = String(p.lab.cep || '').replace(/[^0-9A-Za-z]/g, '').toUpperCase();
  if (cep.length > 8) avisos.push('O CEP do laboratório tem mais de 8 caracteres.');
  return { avisos, erros, tel, cep, cid: cidadeDe(p.lab.cidade), linhas: linhasMaterial(p.materiais) };
}

}
